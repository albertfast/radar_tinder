with corridor as (
  select st_buffer(
    st_geomfromtext(
      'LINESTRING(-122.4194 37.7749,-122.3255 37.5630,-122.1430 37.4419,-121.8863 37.3382)',
      4326
    )::geography,
    5000
  ) as geom
)
select
  er.source,
  er.type,
  count(*) as total
from public.external_radars er, corridor c
where st_intersects(er.location, c.geom)
group by er.source, er.type
order by total desc;
